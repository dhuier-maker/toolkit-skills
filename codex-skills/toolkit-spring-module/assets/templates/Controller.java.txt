package com.funo.project.{module-name}.controller;

import com.cmiot.common.authentication.context.LoginUserContextHolder;
import com.cmiot.common.authority.apis.ApiResource;
import com.cmiot.common.authority.apis.ResourceScope;
import com.cmiot.common.model.Result;
import com.cmiot.common.psk.tenant.context.TenantContextHolder;
import com.cmiot.common.utils.IdGenerator;
import com.funo.project.{module-name}.entity.{business}.{EntityName};
import com.funo.project.{module-name}.entity.{business}.{EntityName}DTO;
import com.funo.project.{module-name}.service.{business}.I{EntityName}Service;
import org.springframework.beans.BeanUtils;
import org.springframework.web.bind.annotation.*;

import javax.annotation.Resource;
import javax.validation.Valid;
import java.util.Date;

/**
 * {description} 前端控制器
 *
 * @author author
 */
@RestController
@RequestMapping("/{entity-name}")
@ApiResource(menu = "{module-name}", value = ResourceScope.AUTHENTICATED)
public class {EntityName}Controller {

    @Resource
    private I{EntityName}Service {entity-name}Service;

    /**
     * 列表查询
     * @return 列表
     */
    @PostMapping("/list")
    public Result getList() {
        return Result.succeed({entity-name}Service.getList());
    }

    /**
     * 详情查询
     * @param id 主键
     * @return 详情
     */
    @GetMapping("/detail/{id}")
    public Result getDetail(@PathVariable("id") Long id) {
        return Result.succeed({entity-name}Service.getById(id));
    }

    /**
     * 新增
     * @param dto 数据传输对象
     * @return 结果
     */
    @PostMapping("/add")
    public Result add(@Valid @RequestBody {EntityName}DTO dto) {
        {EntityName} entity = new {EntityName}();
        BeanUtils.copyProperties(dto, entity);
        entity.setId(IdGenerator.getId());
        entity.setTenantId(TenantContextHolder.getTenant());
        entity.setCreateBy(LoginUserContextHolder.getUserId());
        entity.setCreateTime(new Date());
        boolean result = {entity-name}Service.save{EntityName}(entity);
        return result ? Result.success() : Result.failure("新增失败");
    }

    /**
     * 修改
     * @param dto 数据传输对象
     * @return 结果
     */
    @PostMapping("/update")
    public Result update(@Valid @RequestBody {EntityName}DTO dto) {
        if (dto.getId() == null) {
            return Result.failure("主键不能为空");
        }
        {EntityName} entity = new {EntityName}();
        BeanUtils.copyProperties(dto, entity);
        entity.setUpdateBy(LoginUserContextHolder.getUserId());
        entity.setUpdateTime(new Date());
        boolean result = {entity-name}Service.update{EntityName}(entity);
        return result ? Result.success() : Result.failure("修改失败");
    }

    /**
     * 删除
     * @param id 主键
     * @return 结果
     */
    @GetMapping("/delete/{id}")
    public Result delete(@PathVariable("id") Long id) {
        boolean result = {entity-name}Service.delete{EntityName}(id);
        return result ? Result.success() : Result.failure("删除失败");
    }

}
