package com.funo.project.{module-name}.feign;

import com.cmiot.common.model.Result;
import org.springframework.cloud.openfeign.FeignClient;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestParam;

import java.util.List;

/**
 * {description} Feign客户端
 *
 * @author author
 */
@FeignClient("{service-name}")
public interface I{EntityName}FeignClient {

    /**
     * 根据ID查询
     * @param id 主键
     * @return 结果
     */
    @GetMapping("/{entity-name}/detail")
    Result<Object> getById(@RequestParam("id") Long id);

    /**
     * 列表查询
     * @return 结果
     */
    @GetMapping("/{entity-name}/list")
    Result<List<Object>> getList();

}
